cmake_minimum_required(VERSION 3.22)
project(kobuild_linux_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.3 EXACT REQUIRED COMPONENTS Core Gui GuiPrivate Widgets Network Sql Test BundledLibpng)
qt_standard_project_setup(REQUIRES 6.11)
qt_add_executable(check main.cpp)
target_link_libraries(check PRIVATE Qt6::Widgets Qt6::GuiPrivate Qt6::Network Qt6::Sql Qt6::Test Qt6::BundledLibpng)
qt_add_resources(check messages PREFIX "/kobuild" FILES message.txt)
